+fn coalesce(allocator: std.mem.Allocator, original: p.Program) data.admission.Error!p.Program {
+    const incoming = try allocator.alloc(usize, original.blocks.len);
+    const entry = try allocator.alloc(bool, original.blocks.len);
+    const removed = try allocator.alloc(bool, original.blocks.len);
+    @memset(incoming, 0);

+        .branch => |branch| {

+            incoming[@intCast(branch.when_false.block)] +|= 1;
+        },
+        .switch_variant => |selected| {
+            for (selected.cases) |edge| incoming[@intCast(edge.block)] +|= 1;
+        },
+        .unpack_product => |unpack| {
+            incoming[@intCast(unpack.block)] +|= 1;
