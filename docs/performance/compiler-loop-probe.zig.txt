const std = @import("std");
const boundary = @import("boundary");

fn compile(allocator: std.mem.Allocator) ![32]u8 {
    var builder = boundary.source.Builder.init(allocator);
    defer builder.deinit();
    const module = try boundary.source.examples.generator(&builder);
    var compiled = try boundary.program.compile(allocator, module);
    defer compiled.deinit();
    const bytes = try allocator.alloc(u8, try boundary.image_v2.encodedLength(compiled.program));
    defer allocator.free(bytes);
    _ = try compiled.encode(allocator, bytes);
    return boundary.data_v2.wire.digest(bytes);
}

pub fn main(init: std.process.Init) !void {
    const expected = try compile(init.gpa);
    for (0..5) |_| _ = try compile(init.gpa);
    const start = std.Io.Clock.awake.now(init.io);
    for (0..100) |_| {
        const actual = try compile(init.gpa);
        if (!std.mem.eql(u8, &expected, &actual)) return error.ChangedImage;
    }
    const ns = start.durationTo(std.Io.Clock.awake.now(init.io)).nanoseconds;
    var buffer: [1024]u8 = undefined;
    var output = std.Io.File.stdout().writer(init.io, &buffer);
    try std.json.Stringify.value(.{ .iterations = 100, .elapsed_ns = ns, .image_sha256 = std.fmt.bytesToHex(expected, .lower) }, .{}, &output.interface);
    try output.interface.flush();
}
